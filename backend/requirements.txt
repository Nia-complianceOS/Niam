# --- auth ------------------------------------------------------------
# Dummy auth is the deliberate choice for the hackathon (ACTION_PLAN D1),
# but POST /auth/signup and /auth/login are real routes: they call
# passlib's CryptContext(schemes=["bcrypt"]), which needs the bcrypt
# backend at first use. bcrypt was listed here and never actually
# installed, so either route would have raised a 500 the moment anyone
# called it. Pinned to 4.0.1, which passlib 1.7.4 handles cleanly.
PyJWT==2.13.0
passlib==1.7.4
bcrypt==4.0.1

# Fernet, for core/crypto.py. Each user's GitHub token is stored on a
# :GithubConnection node in the same Neo4j instance as everything else,
# so it is encrypted before it gets there -- a database backup, an Aura
# snapshot or a leaked NEO4J_PASSWORD would otherwise hand over a `repo`
# scoped credential for every user at once. Not a transitive dependency
# to rely on: PyGithub pulls cryptography in for its App-auth JWT signing
# today, and a version of it that stops doing so would break this import
# with no other warning.
cryptography==50.0.1
